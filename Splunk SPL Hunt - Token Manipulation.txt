```Comments: 1. Note the search window - can schedule this correlation every 5 minutes, search approximately 15 minutes of telemetry for correlation context, and keep this line:
| where ChildTime>=relative_time(now(), "-5m@m")
That means the search can look backward far enough to find the preceding EID 10 access and late-arriving events while only generating detections for SYSTEM children created during the current alert interval. If your Splunk ingestion latency routinely exceeds five minutes, increase both the context window and alert window appropriately..
2. For alert suppression, use the resulting process GUID as the primary deduplication identifier: Computer + NewProcessGuid, rather than PID alone. This prevents the same child from generating repeat notables during overlapping scheduled-search windows and avoids PID-reuse ambiguity.
3. One further production optimization would be to move the process metadata into the Splunk Endpoint.Processes data model / accelerated summaries while retaining raw Sysmon Event 10 for ProcessAccess correlation. That substantially reduces the Event ID 1 workload on large environments without weakening this behavioral logic.```

        DirectParentAccessValid=1,
            ParentResolvedGuid,

        DonorParentAccessValid=1,
            LastTargetAccessSourceGuid,

        DirectParentIsNonService=1,
            ParentResolvedGuid
    )


| eval TokenSourcePID=case(

        DirectParentAccessValid=1,
            tostring(LastSourceAccessTargetPID),

        DonorParentAccessValid=1,
            tostring(LastTargetAccessTargetPID),

        true(),
            ""
    )

| eval TokenSourceImage=case(

        DirectParentAccessValid=1,
            LastSourceAccessTargetImage,

        DonorParentAccessValid=1,
            LastTargetAccessTargetImage,

        true(),
            ""
    )

| eval TokenSourceUser=case(

        DirectParentAccessValid=1,
            LastSourceAccessTargetUser,

        DonorParentAccessValid=1,
            LastTargetAccessTargetUser,

        true(),
            ""
    )

| eval TokenSourceProcessGuid=case(

        DirectParentAccessValid=1,
            LastSourceAccessTargetGuid,

        DonorParentAccessValid=1,
            LastTargetAccessTargetGuid,

        true(),
            ""
    )

| eval GrantedAccess=case(

        DirectParentAccessValid=1,
            LastSourceAccessGranted,

        DonorParentAccessValid=1,
            LastTargetAccessGranted,

        true(),
            ""
    )

| eval AccessTime=case(

        DirectParentAccessValid=1,
            LastSourceAccessTime,

        DonorParentAccessValid=1,
            LastTargetAccessTime,

        true(),
            null()
    )


| eval ConfidenceScore=case(

        DirectParentAccessValid=1,
            90,

        DonorParentAccessValid=1,
            90,

        DirectParentIsNonService=1,
            50
    )


| eval Security4688SupportsTransition=if(
        Security4688Matched=1
        AND Security4688TargetSID="S-1-5-18"
        AND Security4688CreatorSID!="S-1-5-18"
        AND Security4688CreatorSID!="S-1-5-19"
        AND Security4688CreatorSID!="S-1-5-20",
        1,
        0
    )

| eval ConfidenceScore=
    ConfidenceScore
    + if(
        Security4688SupportsTransition=1,
        10,
        0
    )

| eval ConfidenceScore=if(
        ConfidenceScore>100,
        100,
        ConfidenceScore
    )


| eval DetectionReasons=case(

        Relationship="Direct Parent-Child Token Manipulation",

            "SYSTEM child has direct non-service parent "
            . ManipulatorImage
            . " PID "
            . ManipulatorPID
            . "; same parent accessed SYSTEM token donor "
            . TokenSourceImage
            . " PID "
            . TokenSourcePID
            . " within 120 seconds before privileged child creation",

        Relationship="Token Donor Reported As Child Parent",

            "Non-service process "
            . ManipulatorImage
            . " PID "
            . ManipulatorPID
            . " accessed SYSTEM token donor "
            . TokenSourceImage
            . " PID "
            . TokenSourcePID
            . "; resulting SYSTEM process reports donor as parent",

        true(),

            "SYSTEM child was created directly by non-service process "
            . ManipulatorImage
            . " PID "
            . ManipulatorPID
            . "; matching Sysmon Event ID 10 token-donor evidence was not available"
    )


| where ConfidenceScore>=50

| where ChildTime>=relative_time(now(), "-5m@m")


| eval Time=strftime(
        ChildTime,
        "%Y-%m-%d %H:%M:%S"
    )

| eval AccessTime=if(
        isnotnull(AccessTime),
        strftime(
            AccessTime,
            "%Y-%m-%d %H:%M:%S"
        ),
        ""
    )

| eval Security4688CreatorPID=if(
        Security4688Matched=1,
        tostring(Security4688CreatorPID),
        ""
    )

| eval Security4688CreatorImage=if(
        Security4688Matched=1,
        Security4688CreatorImage,
        ""
    )

| eval Security4688CreatorUser=if(
        Security4688Matched=1,
        Security4688CreatorUser,
        ""
    )

| eval Security4688CreatorSID=if(
        Security4688Matched=1,
        Security4688CreatorSID,
        ""
    )

| eval Security4688TargetUser=if(
        Security4688Matched=1,
        Security4688TargetUser,
        ""
    )

| eval Security4688TargetSID=if(
        Security4688Matched=1,
        Security4688TargetSID,
        ""
    )


| rename
    ChildPID AS NewProcessPID
    ChildImage AS NewProcessImage
    ChildCommandLine AS NewProcessCommandLine
    ChildUser AS NewProcessUser
    ChildIntegrity AS NewProcessIntegrity
    ChildGuid AS NewProcessGuid

    ChildParentPID AS ReportedParentPID
    ChildParentImage AS ReportedParentImage
    ChildParentGuid AS ReportedParentGuid


| table
    Time
    Computer
    ConfidenceScore
    Relationship

    ManipulatorPID
    ManipulatorImage
    ManipulatorUser
    ManipulatorCommandLine
    ManipulatorProcessGuid

    TokenSourcePID
    TokenSourceImage
    TokenSourceUser
    TokenSourceProcessGuid
    GrantedAccess
    AccessTime

    NewProcessPID
    NewProcessImage
    NewProcessCommandLine
    NewProcessUser
    NewProcessIntegrity
    NewProcessGuid

    ReportedParentPID
    ReportedParentImage
    ReportedParentGuid

    Security4688Matched
    Security4688SupportsTransition
    Security4688CreatorPID
    Security4688CreatorImage
    Security4688CreatorUser
    Security4688CreatorSID
    Security4688TargetUser
    Security4688TargetSID

    DetectionReasons

| sort 0 -ConfidenceScore -Time
